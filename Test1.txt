# Bước 9: Kết thúc hành trình & Kiểm tra thuật toán phân tích hành vi tiêu hao (Anomaly)
mutation EndTrip {
  endTrip(
    trip_id: "3",
    distance_km: 25.5,
    is_heavy_traffic: true,
    end_lat: 10.787501,
    end_lon: 106.728739,
    road_condition: "Rough",
    braking_type: "Hard",
    avg_speed_kmh: 18.0,
    
    # 🌟 HƯỚNG DẪN CẤU HÌNH MẪU GIA TỐC (MEMBER/ADMIN TỰ CHỈNH TRÊN SANDBOX ĐỂ PHỤC VỤ KIỂM THỬ):
    # ------------------------------------------------------------------------------------------
    # 🔹 CHẾ ĐỘ 1: "Normal" hoặc "Constant" hoặc "Uniform_Accelerating" -> Lái an toàn
    #    -> Hệ số phạt tiêu thụ thấp, lượng xăng đốt nằm trong ngưỡng lý thuyết của Winner X.
    #    -> HỆ THỐNG TRẢ VỀ: has_anomaly: false (Trang Web Admin sẽ hiển thị nhãn "An toàn" màu xanh).
    #
    # 🔹 CHẾ ĐỘ 2: "Aggressive" hoặc "Accelerating" -> Lái gắt, phóng nhanh phanh gấp
    #    -> Bộ não `speedBehaviorAnalyzer` phạt x1.25 xăng, đẩy tổng tiêu thụ vượt ngưỡng an toàn 60%.
    #    -> HỆ THỐNG TRẢ VỀ: has_anomaly: true (Trang Web Admin hiển thị nút "⚠️ Bất thường" màu đỏ).
    #
    # 💡 LƯU Ý THỰC TẾ: Trên ứng dụng Flutter Mobile chạy ngoài đường, trường này sẽ do thuật toán Front-End 
    #    tự suy ra từ biến thiên vận tốc GPS từng giây của cảm biến máy rồi tự bắn lên, người dùng không tự chọn.
    acceleration_pattern: "Aggressive" 
  ) {
    id
    status
    total_distance_km
    estimated_fuel_liters
    has_anomaly             # Sẽ trả về True/False bám sát theo cấu hình chuỗi pattern bạn chọn ở trên
  }
}
